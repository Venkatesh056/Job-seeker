# AI Career Agent — backend runtime dependencies
#
# Python 3.12 or 3.13 required.
#   numpy 2.5 needs >=3.12; litellm caps at <3.15.
#
# Install:
#   python -m venv .venv
#   .venv\Scripts\activate           (Windows)     |  source .venv/bin/activate  (POSIX)
#   pip install -r requirements.txt
#   playwright install chromium      <-- required; downloads the browser binary
#
# Dev/test tooling lives in requirements-dev.txt.
# Frontend (Next.js) dependencies live in web/package.json, added in Phase 0.
#
# Versions are exact pins for reproducibility. Upgrade deliberately, not incidentally.

# ─── Agent orchestration ────────────────────────────────────────────────────
# LangGraph runs the multi-agent campaign: Send() fans a region out across sourcing
# agents, then across per-company enrichment, and fans back in. That map-reduce over a
# shared state is what a graph is genuinely better at than hand-rolled asyncio.
#
# The live browser page held across an approval still lives outside the graph, in
# `applications/sessions.py` — a page cannot be serialized into a checkpointer, and
# pretending otherwise would mean approving a form the user never saw.
langgraph==1.2.11
langgraph-checkpoint-postgres==3.1.2

# ─── LLM access ─────────────────────────────────────────────────────────────
# One SDK for OpenAI / Anthropic / Gemini / Groq / xAI / Ollama. Our own LLMRouter
# sits on top and owns routing, caching, fallback chains and budget enforcement.
litellm==1.100.1
tiktoken==0.14.0                  # token counting for per-application cost reporting

# ─── Embeddings (local option) ──────────────────────────────────────────────
# ONNX-based, no torch. Lets resume/JD embeddings run entirely on-device when the
# user prefers no third-party egress. Provider embeddings go through litellm.
fastembed==0.8.0
numpy==2.5.3

# ─── HTTP + resilience ──────────────────────────────────────────────────────

# ─── HTML parsing ───────────────────────────────────────────────────────────
